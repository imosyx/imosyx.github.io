export default { async fetch(request, env) { const url = new URL(request.url); const ip = request.headers.get("CF-Connecting-IP"); // ✅ Pulled from Workers secrets — never in source code const ADMIN_USER = env.ADMIN_USER; const ADMIN_PASS = env.ADMIN_PASS; const CLIENT_SECRET = env.CLIENT_SECRET; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS" }; if (request.method === "OPTIONS") { return new Response(null, { headers: cors }); } // Rate limit helper — simple per-IP counter in KV async function checkRateLimit(key, max, windowSecs) { const now = Math.floor(Date.now() / 1000); const bucket = Math.floor(now / windowSecs); const kvKey = `rl_${key}_${bucket}`; const raw = await env.DB.get(kvKey); const count = raw ? parseInt(raw) : 0; if (count >= max) return false; await env.DB.put(kvKey, String(count + 1), { expirationTtl: windowSecs * 2 }); return true; } // LOGIN — rate limited to 10 attempts per IP per 5 minutes if (url.pathname === "/login" && request.method === "POST") { const allowed = await checkRateLimit(`login_${ip}`, 10, 300); if (!allowed) { return new Response("Too many attempts", { status: 429, headers: cors }); } const body = await request.json(); if (body.user === ADMIN_USER && body.pass === ADMIN_PASS) { const token = crypto.randomUUID(); await env.DB.put("session_" + token, JSON.stringify({ ip }), { expirationTtl: 60 * 60 * 8 // sessions expire after 8 hours automatically }); return new Response(JSON.stringify({ token }), { headers: { ...cors, "Content-Type": "application/json" } }); } return new Response("Invalid login", { status: 401, headers: cors }); } // CLIENT VALIDATE — rate limited to 60 per IP per minute if (url.pathname === "/validate" && request.method === "POST") { const secret = request.headers.get("x-client-secret"); if (!secret || secret !== CLIENT_SECRET) { return new Response("Forbidden", { status: 403, headers: cors }); } const allowed = await checkRateLimit(`validate_${ip}`, 60, 60); if (!allowed) { return new Response("Too many requests", { status: 429, headers: cors }); } let data = JSON.parse(await env.DB.get("keys") || '{"keys":[]}'); const body = await request.json(); let k = data.keys.find(x => x.key === body.key); if (!k) return new Response(JSON.stringify({ valid: false }), { headers: cors }); if (k.revoked || k.disabled || (k.expires && new Date(k.expires) < new Date())) { return new Response(JSON.stringify({ valid: false }), { headers: cors }); } return new Response(JSON.stringify({ valid: true, owner: k.owner, tags: k.tags || [] }), { headers: { ...cors, "Content-Type": "application/json" } }); } // AUTH — all routes below require a valid session const token = request.headers.get("Authorization"); if (!token) return new Response("Unauthorized", { status: 401, headers: cors }); let sessionRaw = await env.DB.get("session_" + token); if (!sessionRaw) return new Response("Unauthorized", { status: 401, headers: cors }); let session = JSON.parse(sessionRaw); if (session.ip !== ip) { return new Response("IP mismatch", { status: 403, headers: cors }); } // ✅ Refresh session TTL on every authenticated request (sliding expiry) await env.DB.put("session_" + token, sessionRaw, { expirationTtl: 60 * 60 * 8 }); let data = JSON.parse(await env.DB.get("keys") || '{"keys":[]}'); // Auto-expire data.keys.forEach(k => { if (k.expires && new Date(k.expires) < new Date()) { k.revoked = true; } }); // GET if (url.pathname === "/get") { return new Response(JSON.stringify(data), { headers: { ...cors, "Content-Type": "application/json" } }); } // ADD if (url.pathname === "/add" && request.method === "POST") { const body = await request.json(); if (!body.key) return new Response("Key is required", { status: 400, headers: cors }); data.keys.push({ id: crypto.randomUUID(), owner: body.owner || "", key: body.key, created: new Date().toISOString(), expires: body.expires || null, revoked: false, disabled: false, tags: Array.isArray(body.tags) ? body.tags : [], notes: body.notes || "" }); await env.DB.put("keys", JSON.stringify(data)); return new Response("OK", { headers: cors }); } // REVOKE if (url.pathname === "/revoke" && request.method === "POST") { const body = await request.json(); let k = data.keys.find(x => x.id === body.id); if (!k) return new Response("Not found", { status: 404, headers: cors }); k.revoked = true; k.revokedAt = new Date().toISOString(); await env.DB.put("keys", JSON.stringify(data)); return new Response("OK", { headers: cors }); } // DELETE if (url.pathname === "/delete" && request.method === "POST") { const body = await request.json(); data.keys = data.keys.filter(x => x.id !== body.id); await env.DB.put("keys", JSON.stringify(data)); return new Response("OK", { headers: cors }); } // TOGGLE if (url.pathname === "/toggle" && request.method === "POST") { const body = await request.json(); let k = data.keys.find(x => x.id === body.id); if (!k) return new Response("Not found", { status: 404, headers: cors }); k.disabled = !k.disabled; k.disabledAt = k.disabled ? new Date().toISOString() : null; await env.DB.put("keys", JSON.stringify(data)); return new Response("OK", { headers: cors }); } // ROTATE if (url.pathname === "/rotate" && request.method === "POST") { const body = await request.json(); if (!body.newKey) return new Response("newKey is required", { status: 400, headers: cors }); let k = data.keys.find(x => x.id === body.id); if (!k) return new Response("Not found", { status: 404, headers: cors }); k.previousKey = k.key; k.key = body.newKey; k.rotatedAt = new Date().toISOString(); await env.DB.put("keys", JSON.stringify(data)); return new Response("OK", { headers: cors }); } // UPDATE if (url.pathname === "/update" && request.method === "POST") { const body = await request.json(); let k = data.keys.find(x => x.id === body.id); if (!k) return new Response("Not found", { status: 404, headers: cors }); if (body.owner !== undefined) k.owner = body.owner; if (body.notes !== undefined) k.notes = body.notes; if (body.tags !== undefined) k.tags = Array.isArray(body.tags) ? body.tags : []; if (body.expires !== undefined) k.expires = body.expires || null; k.updatedAt = new Date().toISOString(); await env.DB.put("keys", JSON.stringify(data)); return new Response("OK", { headers: cors }); } return new Response("Not found", { status: 404, headers: cors }); } };
